@startuml

  :Отображение сообщения об ошибке;
  stop
endif
:Проверка уникальности логина в БД;

--- Рисунок 2.7 - Блок-схема перевода средств ---

@startuml
start
:Пользователь указывает счёт списания,\nполучателя и сумму;
:Проверка существования счёта списания;
if (Счёт найден?) then (нет)
  :Ошибка "Счёт не найден";
  stop
endif
:Проверка прав доступа пользователя к счёту;
if (Доступ разрешён?) then (нет)
  :Ошибка "Нет прав на счёт";
  stop
endif
:Проверка статуса блокировки счёта списания;
if (Счёт заблокирован?) then (да)
  :Ошибка "Счёт заблокирован";
  stop
endif
:Поиск счёта получателя;
if (Счёт получателя найден?) then (нет)
  :Ошибка "Счёт получателя не найден";
  stop
endif
if (Получатель — юридическое лицо?) then (да)
  :Проверка совпадения УНП;
  if (УНП совпадает?) then (нет)
    :Ошибка "УНП не совпадает";
    stop
  endif
endif
:Проверка совпадения валют счетов;
if (Валюты совпадают?) then (нет)
  :Ошибка "Валюты не совпадают";
  stop
endif
:Проверка достаточности средств;
if (Средств достаточно?) then (нет)
  :Ошибка "Недостаточно средств";
  stop
endif
:Списание суммы со счёта отправителя;
:Зачисление суммы на счёт получателя;
:Создание записи об операции в БД;
:Возврат данных операции с идентификатором;
stop
@enduml


--- Рисунок 2.8 - Блок-схема подписания платёжного поручения ЭЦП ---

@startuml
start
:Пользователь нажимает "Подписать ЭЦП";
:Проверка наличия приватного ключа\nв localStorage браузера;
if (Ключ найден?) then (нет)
  :Ошибка "Ключ ЭЦП не найден";
  stop
endif
:Формирование payload:\n"id|сумма|получатель|счёт|назначение";
:Подписание payload приватным ключом\n(RSASSA-PKCS1-v1_5, SHA-256);
:Кодирование подписи в Base64;
:Отправка подписи на сервер;
:Сервер извлекает публичный ключ\nорганизации из БД;
if (Публичный ключ найден?) then (нет)
  :Проверка длины подписи (>= 10 символов);
else (да)
  :Формирование идентичного payload\nна основе данных поручения из БД;
  :Верификация подписи (RSA.VerifyData);
  if (Подпись действительна?) then (нет)
    :Ошибка "Подпись ЭЦП недействительна";
    stop
  endif
endif
:Обновление статуса поручения на "Подписан";
:Сохранение значения подписи в БД;
:Возврат обновлённых данных поручения;
stop
@enduml
